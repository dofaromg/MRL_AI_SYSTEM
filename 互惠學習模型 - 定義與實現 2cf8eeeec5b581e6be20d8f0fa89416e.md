# 互惠學習模型 - 定義與實現

## 概述

互惠學習模型是神經共生網絡的關鍵組件，負責管理網絡間的知識和參數共享機制。通過此模型，不同的神經網絡能夠從彼此的學習經驗中獲益，實現經驗交換與集體智慧，從而大幅提高學習效率和泛化能力。

## 理論基礎

### 互惠學習理論

互惠學習理論源於生物學中的互惠共生概念，即不同物種通過互惠關係獲得共同收益。在神經網絡領域，這一概念轉化為不同神經網絡通過知識交換來改進性能。

### 協同過濾原理

借鑒協同過濾技術，互惠學習模型能夠根據網絡間的相似性和互補性來有效篩選和分享最相關的知識。

### 知識蒸餾理論

基於知識蒸餾理論，互惠學習模型實現了從一個"教師"網絡到另一個"學生"網絡的知識轉移，使得複雜網絡學到的知識能夠有效地傳遞給更輕量級的網絡。

```python
class ReciprocalLearner:
    def __init__(self, config):
        self.learning_rate = config.get('learning_rate', 0.01)
        self.knowledge_retention = config.get('knowledge_retention', 0.9)

    def learn_from_others(self, learner_network, teacher_networks, context=None):

        if not teacher_networks:
            return {'success': False, 'reason': 'No teacher networks provided'}
            
        learning_results = []
        
        for teacher in teacher_networks:
            # 評估與教師網絡的相容性
            compatibility = self._assess_compatibility(learner_network, teacher)
            
            # 如果相容性太低，跳過此教師
            if compatibility < 0.3:  # 最低相容性閾值
                learning_results.append({
                    'teacher_id': [teacher.id](http://teacher.id),
                    'success': False,
                    'reason': 'Low compatibility',
                    'compatibility_score': compatibility
                })
                continue
            
            # 從教師網絡獲取知識
            knowledge = self._extract_knowledge(teacher)
            
            # 應用知識過濾器
            for filter_fn in self.knowledge_filters:
                knowledge, filter_meta = filter_fn(knowledge, teacher, learner_network)
                
                # 如果過濾後知識為空，跳過此教師
                if not knowledge:
                    learning_results.append({
                        'teacher_id': [teacher.id](http://teacher.id),
                        'success': False,
                        'reason': 'Knowledge filtered out',
                        'filter_meta': filter_meta
                    })
                    continue
            
            # 整合知識到學習者網絡
            integration_result = self._integrate_knowledge(learner_network, knowledge, compatibility)
            
            learning_results.append({
                'teacher_id': [teacher.id](http://teacher.id),
                'success': integration_result['success'],
                'compatibility_score': compatibility,
                'knowledge_integrated': integration_result.get('integrated_items', 0),
                'learning_impact': integration_result.get('impact', 0.0)
            })
            
        return {
            'success': any(result['success'] for result in learning_results),
            'results': learning_results,
            'networks_learned_from': sum(1 for result in learning_results if result['success'])
        }
    
    def register_knowledge_filter(self, filter_function):
        """註冊知識過濾器"""
        self.knowledge_filters.append(filter_function)
        return len(self.knowledge_filters)
    
    def _assess_compatibility(self, network_a, network_b):
        """評估兩個網絡的相容性"""
        # 基本架構相容性
        structural_compatibility = self._structural_compatibility(network_a, network_b)
        
        # 功能相容性
        functional_compatibility = self._functional_compatibility(network_a, network_b)
        
        # 如果啟用了共振敏感性，考慮共振相容性
        if self.resonance_sensitivity and hasattr(network_a, 'get_resonance_state') and hasattr(network_b, 'get_resonance_state'):
            resonance_a = network_a.get_resonance_state()
            resonance_b = network_b.get_resonance_state()
            resonance_compatibility = self._calculate_resonance_compatibility(resonance_a, resonance_b)
            
            # 結合三種相容性
            return 0.4 * structural_compatibility + 0.4 * functional_compatibility + 0.2 * resonance_compatibility
        else:
            # 僅結合結構和功能相容性
            return 0.5 * structural_compatibility + 0.5 * functional_compatibility
    
    def _integrate_knowledge(self, target_network, knowledge, compatibility_score):
        """將知識整合到目標網絡"""
        if self.integration_strategy == 'weighted_average':
            return self._weighted_average_integration(target_network, knowledge, compatibility_score)
        elif self.integration_strategy == 'selective_update':
            return self._selective_update_integration(target_network, knowledge, compatibility_score)
        elif self.integration_strategy == 'adaptive_fusion':
            return self._adaptive_fusion_integration(target_network, knowledge, compatibility_score)
        else:
            # 默認使用加權平均整合
            return self._weighted_average_integration(target_network, knowledge, compatibility_score)
```

### 3. 經驗記憶庫

存儲和管理學習經驗，用於提高學習效率：

```python
class ExperienceMemory:
    def __init__(self, config):
        self.memory_capacity = config.get('capacity', 1000)
        self.importance_threshold = config.get('importance_threshold', 0.5)
        self.consolidation_interval = config.get('consolidation_interval', 100)
        self.experiences = []
        self.experience_index = {}  # 經驗索引，用於快速查找
        self.meta_memories = {}  # 元記憶，存儲關於經驗的統計信息
        self.last_consolidation = 0  # 上次記憶整合的時間
        
    def add_experience(self, experience_data):
        """添加新的學習經驗"""
        # 計算經驗的重要性
        importance = self._calculate_importance(experience_data)
        
        # 如果經驗太不重要且記憶已滿，則忽略
        if importance < self.importance_threshold and len(self.experiences) >= self.memory_capacity:
            return False
            
        # 創建經驗記錄
        experience_id = str(uuid.uuid4())
        experience_record = {
            'id': experience_id,
            'timestamp': time.time(),
            'data': experience_data,
            'importance': importance,
            'access_count': 0,
            'last_access': time.time()
        }
        
        # 如果記憶已滿，移除最不重要的經驗
        if len(self.experiences) >= self.memory_capacity:
            self._remove_least_important_experience()
            
        # 添加新經驗
        self.experiences.append(experience_record)
        
        # 更新索引
        self._update_index(experience_record)
        
        # 更新元記憶
        self._update_meta_memories(experience_record)
        
        # 檢查是否需要記憶整合
        if len(self.experiences) - self.last_consolidation >= self.consolidation_interval:
            self._consolidate_memories()
            
        return experience_id
        
    def retrieve_experience(self, query, top_k=5):
        """檢索相關經驗"""
        # 計算查詢與所有經驗的相關性
        relevance_scores = []
        for exp in self.experiences:
            relevance = self._calculate_relevance(query, exp['data'])
            relevance_scores.append((exp, relevance))
            
        # 按相關性排序
        relevance_scores.sort(key=lambda x: x[1], reverse=True)
        
        # 返回最相關的 top_k 個經驗
        results = []
        for exp, score in relevance_scores[:top_k]:
            # 更新訪問信息
            exp['access_count'] += 1
            exp['last_access'] = time.time()
            
            # 添加到結果列表
            results.append({
                'experience': exp['data'],
                'relevance': score,
                'importance': exp['importance'],
                'id': exp['id'],
                'timestamp': exp['timestamp']
            })
            
        return results
        
    def record_exchange(self, source_id, target_id, exchange_result):
        """記錄交換經驗"""
        exchange_data = {
            'type': 'exchange',
            'source_id': source_id,
            'target_id': target_id,
            'success': exchange_result.get('success', False),
            'knowledge_items': exchange_result.get('knowledge_items', 0),
            'impact': exchange_result.get('impact', 0.0)
        }
        
        return self.add_experience(exchange_data)
        
    def _calculate_importance(self, experience_data):
        """計算經驗的重要性"""
        # 基本重要性計算可以考慮多種因素
        importance = 0.5  # 默認中等重要性
        
        # 如果是成功的知識交換，重要性提高
        if experience_data.get('type') == 'exchange' and experience_data.get('success', False):
            importance += 0.2
            
            # 知識影響越大，重要性越高
            impact = experience_data.get('impact', 0.0)
            importance += 0.3 * impact
            
        # 檢查是否與現有元記憶匹配
        for meta_type, meta_data in self.meta_memories.items():
            if self._matches_meta_pattern(experience_data, meta_type, meta_data):
                importance += 0.1  # 與元模式匹配的經驗更重要
                
        return min(1.0, importance)  # 確保重要性不超過1.0
        
    def _remove_least_important_experience(self):
        """移除最不重要的經驗"""
        if not self.experiences:
            return
            
        # 計算每個經驗的綜合重要性（結合固有重要性和訪問頻率）
        composite_scores = []
        for i, exp in enumerate(self.experiences):
            # 時間衰減因子
            time_factor = math.exp(-0.01 * (time.time() - exp['timestamp']) / (60 * 60 * 24))  # 每天衰減約1%
            
            # 訪問頻率因子
            access_factor = 1.0 + 0.1 * min(10, exp['access_count'])  # 最多增加100%
            
            # 綜合分數
            score = exp['importance'] * time_factor * access_factor
            composite_scores.append((i, score))
            
        # 找到分數最低的經驗
        min_idx, min_score = min(composite_scores, key=lambda x: x[1])
        
        # 從索引中移除
        removed_exp = self.experiences[min_idx]
        self._remove_from_index(removed_exp)
        
        # 從經驗列表中移除
        del self.experiences[min_idx]
```

### 4. 知識轉換器

實現不同網絡間知識表示的轉換：

```python
class KnowledgeTransformer:
    def __init__(self, config):
        self.transformation_strategies = {
            'direct': self._direct_transformation,
            'scaled': self._scaled_transformation,
            'structural': self._structural_transformation,
            'functional': self._functional_transformation
        }
        self.default_strategy = config.get('default_strategy', 'direct')
        self.adaptation_rate = config.get('adaptation_rate', 0.1)
        self.transformation_cache = {}  # 緩存常用的轉換
        
    def transform_knowledge(self, knowledge, source_network, target_network, strategy=None):
        """將知識從源網絡格式轉換為目標網絡格式"""
        if strategy is None:
            strategy = self.default_strategy
            
        if strategy not in self.transformation_strategies:
            raise ValueError(f"Unknown transformation strategy: {strategy}")
            
        # 檢查緩存
        cache_key = (source_[network.id](http://network.id), target_[network.id](http://network.id), strategy)
        if cache_key in self.transformation_cache:
            transformer = self.transformation_cache[cache_key]
        else:
            # 創建新的轉換函數
            transformer = self._create_transformer(source_network, target_network, strategy)
            self.transformation_cache[cache_key] = transformer
            
        # 應用轉換
        try:
            transformed_knowledge = transformer(knowledge)
            return {
                'success': True,
                'transformed_knowledge': transformed_knowledge,
                'strategy': strategy
            }
        except Exception as e:
            return {
                'success': False,
                'error': str(e),
                'strategy': strategy
            }
            
    def _create_transformer(self, source_network, target_network, strategy):
        """創建從源網絡到目標網絡的轉換函數"""
        # 獲取策略函數
        transform_func = self.transformation_strategies[strategy]
        
        # 獲取網絡結構信息
        source_structure = self._get_network_structure(source_network)
        target_structure = self._get_network_structure(target_network)
        
        # 創建映射
        mapping = self._create_mapping(source_structure, target_structure, strategy)
        
        # 返回帶有預設參數的轉換函數
        def transformer(knowledge):
            return transform_func(knowledge, mapping, source_network, target_network)
        
        return transformer
```

```python
def assess_network_compatibility(network_a, network_b):
    """評估兩個網絡的相容性"""
    compatibility_scores = {}
    
    # 1. 架構相容性
    architectural_compatibility = calculate_architectural_compatibility(network_a, network_b)
    compatibility_scores['architectural'] = architectural_compatibility
    
    # 2. 任務相容性
    task_compatibility = calculate_task_compatibility(network_a, network_b)
    compatibility_scores['task'] = task_compatibility
    
    # 3. 表示相容性
    representation_compatibility = calculate_representation_compatibility(network_a, network_b)
    compatibility_scores['representation'] = representation_compatibility
    
    # 4. 共振相容性 (如果適用)
    if hasattr(network_a, 'resonance_layers') and hasattr(network_b, 'resonance_layers'):
        resonance_compatibility = calculate_resonance_compatibility(
            network_a.resonance_layers, 
            network_b.resonance_layers
        )
        compatibility_scores['resonance'] = resonance_compatibility
    
    # 計算綜合相容性
    weights = {
        'architectural': 0.3,
        'task': 0.3,
        'representation': 0.3,
        'resonance': 0.1
    }
    
    overall_compatibility = sum(
        score * weights.get(key, 0.0) 
        for key, score in compatibility_scores.items()
    ) / sum(
        weights.get(key, 0.0) 
        for key in compatibility_scores.keys()
    )
    
    compatibility_scores['overall'] = overall_compatibility
    
    return compatibility_scores
```

```python
    }
```

```python
    """加權整合知識"""
    # 獲取網絡結構
    network_structure = get_network_structure(network)
    
    # 跟踪哪些部分被更新

        if layer_name in network_structure['layers']:
            layer = network_structure['layers'][layer_name]
            
            # 整合權重 (加權平均)

                original_weight = 0.7  # 原始權重的權重
                new_weight = 0.3  # 新知識的權重

                        layer['weights'][param_name] = (
                            original_weight * layer['weights'][param_name] + 
                            new_weight * param_value
                        )
                        
                        # 記錄更新
                        if layer_name not in updated_components:
                            updated_components[layer_name] = {}
                        if 'weights' not in updated_components[layer_name]:
                            updated_components[layer_name]['weights'] = []
                            
                        updated_components[layer_name]['weights'].append(param_name)
    
    # 更新網絡結構
    update_network_structure(network, network_structure)
    
    # 返回整合結果
    return {
        'success': True,
        'updated_components': updated_components,
        'integration_method': 'weighted',
        'components_count': count_updated_components(updated_components)
    }
```

```python
class AdaptiveLearningStrategy:
    def __init__(self, config):
        self.base_learning_rate = config.get('base_learning_rate', 0.01)
        self.min_learning_rate = config.get('min_learning_rate', 0.001)
        self.max_learning_rate = config.get('max_learning_rate', 0.1)
        self.performance_threshold = config.get('performance_threshold', 0.7)
        self.adaptation_factor = config.get('adaptation_factor', 0.1)
        
    def adjust_learning_parameters(self, network, performance_metrics):
        """根據性能指標調整學習參數"""
        current_performance = performance_metrics.get('accuracy', 0.5)
        
        # 基於當前性能調整學習率
        if current_performance > self.performance_threshold:
            # 性能良好，微調學習參數
            learning_rate = self.base_learning_rate * (1.0 - self.adaptation_factor)
        else:
            # 性能不佳，增大學習率促進學習
            performance_gap = self.performance_threshold - current_performance
            learning_rate = self.base_learning_rate * (1.0 + self.adaptation_factor * performance_gap)
            
        # 確保學習率在合理範圍內
        learning_rate = max(self.min_learning_rate, min(learning_rate, self.max_learning_rate))
        
        # 返回調整後的參數
        return {
            'learning_rate': learning_rate,
            'momentum': self._adjust_momentum(current_performance),
            'regularization': self._adjust_regularization(current_performance)
        }
        
    def select_teacher_networks(self, learner_network, candidate_teachers, performance_metrics):
        """選擇最適合的教師網絡"""
        current_performance = performance_metrics.get('accuracy', 0.5)
        selected_teachers = []
        
        # 計算與候選教師的相容性
        compatibility_scores = {}
        for teacher in candidate_teachers:
            compatibility = calculate_network_compatibility(learner_network, teacher)
            compatibility_scores[[teacher.id](http://teacher.id)] = compatibility
            
        # 如果性能良好，選擇相容性高的教師進行微調
        if current_performance > self.performance_threshold:
            # 選擇相容性最高的前3個教師
            sorted_teachers = sorted(
                candidate_teachers, 
                key=lambda t: compatibility_scores[[t.id](http://t.id)],
                reverse=True
            )
            selected_teachers = sorted_teachers[:3]
        else:
            # 性能不佳，同時選擇相容性高和多樣性高的教師
            high_compatibility_teachers = [
                t for t in candidate_teachers 
                if compatibility_scores[[t.id](http://t.id)] > 0.7
            ][:2]
            
            # 選擇一些多樣性高的教師
            diverse_teachers = self._select_diverse_teachers(
                candidate_teachers, 
                high_compatibility_teachers, 
                3
            )
            
            selected_teachers = high_compatibility_teachers + diverse_teachers
```

### 協同學習

多個網絡共同學習同一任務，共享經驗和知識：

```python
class CollaborativeLearningStrategy:
    def __init__(self, config):
        self.collaboration_strength = config.get('strength', 0.5)
        self.synchronization_interval = config.get('sync_interval', 10)
        self.shared_experience_pool = []
        [self.network](http://self.network)_contributions = {}
        self.collaboration_history = []
        self.iteration_counter = 0
        
    def collaborate(self, networks, task_data):
        """讓多個網絡協同學習任務"""
        self.iteration_counter += 1
        
        # 每個網絡獨立學習
        individual_results = {}
        for network in networks:
            result = network.learn(task_data)
            individual_results[[network.id](http://network.id)] = result
            
        # 記錄貢獻
        for network_id, result in individual_results.items():
            if network_id not in [self.network](http://self.network)_contributions:
                [self.network](http://self.network)_contributions[network_id] = 0
                
            # 根據學習性能增加貢獻分數
            [self.network](http://self.network)_contributions[network_id] += result.get('performance_gain', 0.01)
            
        # 收集共享經驗
        new_experiences = []
        for network_id, result in individual_results.items():
            if 'shareable_experience' in result:
                experience = result['shareable_experience']
                experience['source_network_id'] = network_id
                experience['timestamp'] = time.time()
                new_experiences.append(experience)
                
        # 添加到共享經驗池
        self.shared_experience_pool.extend(new_experiences)
        
        # 限制經驗池大小
        if len(self.shared_experience_pool) > 1000:
            # 保留最新和最有價值的經驗
            self.shared_experience_pool = self._prune_experience_pool(1000)
            
        # 檢查是否需要同步
        if self.iteration_counter % self.synchronization_interval == 0:
            # 執行知識同步
            sync_results = self._synchronize_networks(networks)
            
            # 記錄協同歷史
            self.collaboration_history.append({
                'iteration': self.iteration_counter,
                'individual_results': individual_results,
                'sync_results': sync_results,
                'timestamp': time.time()
            })
            
            return {
                'individual_results': individual_results,
                'sync_performed': True,
                'sync_results': sync_results,
                'experiences_shared': len(new_experiences)
            }
            
        return {
            'individual_results': individual_results,
            'sync_performed': False,
            'experiences_shared': len(new_experiences)
        }
        
    def _synchronize_networks(self, networks):
        """同步網絡知識"""
        sync_results = {}
        
        # 計算每個網絡的同步權重 (基於貢獻)
        total_contribution = sum([self.network](http://self.network)_contributions.values())
        if total_contribution == 0:
            # 如果沒有貢獻記錄，使用均勻權重
            network_weights = {[network.id](http://network.id): 1.0 / len(networks) for network in networks}
        else:
            network_weights = {
                network_id: contribution / total_contribution
                for network_id, contribution in [self.network](http://self.network)_contributions.items()
            }
            
        # 對每對網絡執行知識交換
        for i, network_a in enumerate(networks):
            sync_results[network_[a.id](http://a.id)] = {}
            
            for j, network_b in enumerate(networks):
                if i == j:
                    continue  # 跳過自己
                    
                # 計算交換強度
                exchange_strength = self.collaboration_strength * (
                    network_weights.get(network_[a.id](http://a.id), 0.0) +
                    network_weights.get(network_[b.id](http://b.id), 0.0)
                ) / 2.0
                
                # 執行知識交換
                exchange_result = exchange_knowledge(
                    network_a, 
                    network_b, 
                    strength=exchange_strength
                )
                
                sync_results[network_[a.id](http://a.id)][network_[b.id](http://b.id)] = exchange_result
                
        return sync_results
```

### 競爭學習

網絡競爭以獲得更好的性能，只有最優秀的網絡可以分享知識：

```python
class CompetitiveLearningStrategy:
    def __init__(self, config):
        self.competition_rounds = config.get('rounds', 5)
        self.selection_percentage = config.get('selection_percentage', 0.3)
        self.mutation_rate = config.get('mutation_rate', 0.05)
        self.crossover_rate = config.get('crossover_rate', 0.3)
        
    def compete(self, networks, evaluation_data):
        """讓網絡進行競爭學習"""
        round_results = []
        
        for round_idx in range(self.competition_rounds):
            # 評估每個網絡的性能
            performance_scores = {}
            for network in networks:
                score = self._evaluate_network(network, evaluation_data)
                performance_scores[[network.id](http://network.id)] = score
                
            # 選擇表現最好的網絡
            networks_with_scores = [(network, performance_scores[[network.id](http://network.id)]) for network in networks]
            networks_with_scores.sort(key=lambda x: x[1], reverse=True)
            
            # 計算選擇數量
            selection_count = max(1, int(len(networks) * self.selection_percentage))
            selected_networks = [pair[0] for pair in networks_with_scores[:selection_count]]
            
            # 從選中的網絡生成新一代
            next_generation = self._generate_next_generation(selected_networks, len(networks))
            
            # 記錄本輪結果
            round_results.append({
                'round': round_idx + 1,
                'performance_scores': performance_scores,
                'selected_network_ids': [[network.id](http://network.id) for network in selected_networks],
                'best_score': networks_with_scores[0][1] if networks_with_scores else 0.0,
                'average_score': sum(performance_scores.values()) / len(performance_scores) if performance_scores else 0.0
            })
            
            # 更新網絡列表
            networks = next_generation
            
        return {
            'final_networks': networks,
            'round_results': round_results,
            'best_network': networks[0] if networks else None,
            'best_score': round_results[-1]['best_score'] if round_results else 0.0
        }
        
    def _generate_next_generation(self, selected_networks, target_size):
        """從選中的網絡生成新一代"""
        next_generation = selected_networks.copy()  # 保留選中的網絡
        
        # 通過交叉和變異生成新網絡
        while len(next_generation) < target_size:
            # 隨機選擇兩個父網絡
            parent_a, parent_b = random.sample(selected_networks, 2) if len(selected_networks) >= 2 else (selected_networks[0], selected_networks[0])
            
            # 創建子網絡
            child = self._crossover(parent_a, parent_b)
            
            # 變異
            child = self._mutate(child)
            
            # 添加到新一代
            next_generation.append(child)
            
        return next_generation
```
